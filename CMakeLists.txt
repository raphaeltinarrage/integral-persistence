cmake_minimum_required(VERSION 3.19)
project(integral_persistence LANGUAGES CXX)
find_package(Python 3.11 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(nanobind CONFIG REQUIRED)
find_path(GMP_INCLUDE_DIR gmpxx.h REQUIRED)
find_path(FLINT_INCLUDE_DIR flint/fmpz_factor.h REQUIRED)
find_library(GMP_LIBRARY gmp REQUIRED)
find_library(GMPXX_LIBRARY gmpxx REQUIRED)
find_library(FLINT_LIBRARY flint REQUIRED)
nanobind_add_module(_native NOMINSIZE cpp/bindings.cpp cpp/construction.cpp)
target_compile_features(_native PRIVATE cxx_std_17)
target_include_directories(_native PRIVATE cpp/include "${GMP_INCLUDE_DIR}" "${FLINT_INCLUDE_DIR}")
target_link_libraries(_native PRIVATE "${FLINT_LIBRARY}" "${GMPXX_LIBRARY}" "${GMP_LIBRARY}")
install(TARGETS _native LIBRARY DESTINATION integral_persistence)
